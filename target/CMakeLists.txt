cmake_minimum_required(VERSION 3.28)
project(kodanuki CXX)

set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
if(NOT CMAKE_BUILD_TYPE)
	set(CMAKE_BUILD_TYPE Release)
endif()
option(KODANUKI_PORTABLE "Build for any x86-64 with AVX2 rather than this machine" OFF)
if(KODANUKI_PORTABLE)
	add_compile_options(-march=x86-64-v3)
else()
	add_compile_options(-march=native)
endif()
add_compile_options(-ftrivial-auto-var-init=uninitialized -ffp-contract=off --param inline-unit-growth=400 -fno-stack-protector -fno-math-errno -fno-trapping-math -Wall -Wextra -Werror=missing-field-initializers -g)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set_property(GLOBAL PROPERTY JOB_POOLS link=1)
set(CMAKE_JOB_POOL_LINK link)

set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
get_filename_component(root ${CMAKE_SOURCE_DIR} DIRECTORY)
include_directories(${root}/source)

set(OFF "" CACHE STRING "Globs of the plugins built off, such as feature/clouds")
separate_arguments(offGlobs UNIX_COMMAND "${OFF}")
set(offPlugins "")
foreach(glob IN LISTS offGlobs)
	file(GLOB named LIST_DIRECTORIES true RELATIVE ${root}/source/plugins ${root}/source/plugins/${glob})
	if(NOT named)
		message(FATAL_ERROR "OFF: ${glob} names no plugin")
	endif()
	foreach(plugin IN LISTS named)
		file(GLOB pluginSources ${root}/source/plugins/${plugin}/*.cpp)
		set(switched FALSE)
		foreach(source IN LISTS pluginSources)
			file(STRINGS ${source} lines REGEX "pluginBuilt\\(\"${plugin}\"\\)")
			if(lines)
				set(switched TRUE)
			endif()
		endforeach()
		if(NOT switched)
			message(FATAL_ERROR "OFF: ${plugin} cannot be built off")
		endif()
		message(NOTICE "built off: ${plugin}")
		list(APPEND offPlugins ${plugin})
	endforeach()
endforeach()
file(GLOB pluginFolders LIST_DIRECTORIES true RELATIVE ${root}/source/plugins ${root}/source/plugins/*/*)
set(windowPlugins "")
foreach(plugin IN LISTS pluginFolders)
	file(GLOB pluginFiles ${root}/source/plugins/${plugin}/*)
	foreach(file IN LISTS pluginFiles)
		file(STRINGS ${file} lines REGEX "#include <SDL3/")
		if(lines)
			list(APPEND windowPlugins ${plugin})
		endif()
	endforeach()
endforeach()
list(REMOVE_DUPLICATES windowPlugins)
foreach(plugin IN LISTS pluginFolders)
	file(GLOB pluginFiles ${root}/source/plugins/${plugin}/*)
	foreach(window IN LISTS windowPlugins)
		foreach(file IN LISTS pluginFiles)
			file(STRINGS ${file} lines REGEX "#include \"plugins/${window}/")
			if(lines AND NOT plugin IN_LIST windowPlugins)
				list(APPEND windowPlugins ${plugin})
			endif()
		endforeach()
	endforeach()
endforeach()
file(GLOB pluginHeaders CONFIGURE_DEPENDS RELATIVE ${root}/source ${root}/source/plugins/*/*/plugin.h)
set(pluginIncludes "")
set(pluginCalls "")
set(windowIncludes "")
set(windowCalls "")
foreach(header IN LISTS pluginHeaders)
	string(REGEX REPLACE "^plugins/(.*)/plugin.h$" "\\1" plugin ${header})
	if(plugin IN_LIST offPlugins)
		continue()
	endif()
	file(STRINGS ${root}/source/${header} declared REGEX "^Plugin [A-Za-z]+\\(\\);$")
	string(REGEX MATCH "[A-Za-z]+Plugin" function "${declared}")
	string(APPEND windowIncludes "#include \"${header}\"\n")
	string(APPEND windowCalls "\t\t${function}(),\n")
	if(NOT plugin IN_LIST windowPlugins)
		string(APPEND pluginIncludes "#include \"${header}\"\n")
		string(APPEND pluginCalls "\t\t${function}(),\n")
	endif()
endforeach()
set(registry "#include \"core/plugin.h\"\n@includes@\n#include <vector>\n\nnamespace kodanuki {\n\nstd::vector<Plugin> everyPlugin() {\n\treturn {\n@calls@\t};\n}\n\n} // namespace kodanuki\n")
set(includes ${pluginIncludes})
set(calls ${pluginCalls})
string(CONFIGURE "${registry}" content @ONLY)
file(CONFIGURE OUTPUT ${CMAKE_BINARY_DIR}/plugins.cpp CONTENT "${content}")
set(includes ${windowIncludes})
set(calls ${windowCalls})
string(CONFIGURE "${registry}" content @ONLY)
file(CONFIGURE OUTPUT ${CMAKE_BINARY_DIR}/plugins_window.cpp CONTENT "${content}")
file(GLOB_RECURSE everyPluginSource CONFIGURE_DEPENDS ${root}/source/plugins/*.cpp)
file(GLOB sources CONFIGURE_DEPENDS ${root}/source/*/*.cpp)
list(APPEND sources ${everyPluginSource})
foreach(plugin IN LISTS offPlugins)
	list(REMOVE_ITEM sources ${root}/source/plugins/${plugin}/plugin.cpp)
endforeach()
set(windowSources ${sources})
foreach(plugin IN LISTS windowPlugins)
	list(FILTER sources EXCLUDE REGEX "/source/plugins/${plugin}/")
endforeach()
list(APPEND sources ${CMAKE_BINARY_DIR}/plugins.cpp)
list(APPEND windowSources ${CMAKE_BINARY_DIR}/plugins_window.cpp)
add_executable(kodanuki ${root}/source/render.cpp ${sources})

file(GLOB tests CONFIGURE_DEPENDS tests/*.cpp)
if(tests)
	add_executable(kodanuki_tests ${tests} ${sources} ${root}/source/plugins/sink/window/fly.cpp)
endif()

find_package(SDL3 CONFIG)
if(SDL3_FOUND)
	add_executable(kodanuki_view ${root}/source/viewer.cpp ${windowSources})
	target_link_libraries(kodanuki_view PRIVATE SDL3::SDL3)
endif()

add_executable(kodanuki_dump tools/dump.cpp ${sources})
add_executable(kodanuki_probe tools/probe.cpp ${sources})
add_executable(kodanuki_cut_stats tools/cut-stats.cpp tools/spots.cpp ${sources})
add_executable(kodanuki_face_stats tools/face-stats.cpp tools/spots.cpp ${sources})
add_executable(kodanuki_tree_stats tools/tree-stats.cpp ${sources})
add_executable(kodanuki_walk_bench tools/walk-bench.cpp ${sources})
