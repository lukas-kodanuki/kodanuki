cmake_minimum_required(VERSION 3.28)
project(kodanuki CXX)

set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
if(NOT CMAKE_BUILD_TYPE)
	set(CMAKE_BUILD_TYPE Release)
endif()
option(KODANUKI_PORTABLE "Build for any x86-64 with AVX2 rather than this machine" OFF)
if(KODANUKI_PORTABLE)
	add_compile_options(-march=x86-64-v3)
else()
	add_compile_options(-march=native)
endif()
add_compile_options(-ftrivial-auto-var-init=uninitialized -ffp-contract=off --param inline-unit-growth=400 -fno-stack-protector -fno-math-errno -fno-trapping-math -Wall -Wextra -Werror=missing-field-initializers -g)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set_property(GLOBAL PROPERTY JOB_POOLS link=1)
set(CMAKE_JOB_POOL_LINK link)

set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
get_filename_component(root ${CMAKE_SOURCE_DIR} DIRECTORY)
include_directories(${root}/source)

set(OFF "" CACHE STRING "Globs of the extras built off, such as *_biased_*")
separate_arguments(offGlobs UNIX_COMMAND "${OFF}")
foreach(glob IN LISTS offGlobs)
	file(GLOB offExtras LIST_DIRECTORIES true RELATIVE ${root}/source/extras ${root}/source/extras/${glob})
	if(NOT offExtras)
		message(FATAL_ERROR "OFF: ${glob} names no extra")
	endif()
	foreach(extra IN LISTS offExtras)
		string(TOUPPER "KODANUKI_OFF_${extra}" switch)
		file(GLOB headers ${root}/source/extras/${extra}/*.h)
		set(switched FALSE)
		foreach(header IN LISTS headers)
			file(STRINGS ${header} lines REGEX "#ifdef ${switch}$")
			if(lines)
				set(switched TRUE)
			endif()
		endforeach()
		if(NOT switched)
			message(FATAL_ERROR "OFF: ${extra} cannot be built off")
		endif()
		message(NOTICE "built off: ${extra}")
		add_compile_definitions(${switch})
	endforeach()
endforeach()
file(GLOB sources CONFIGURE_DEPENDS ${root}/source/*/*.cpp ${root}/source/extras/*/*.cpp)
list(FILTER sources EXCLUDE REGEX "/source/viewer/")
add_executable(kodanuki ${root}/source/render.cpp ${sources})

file(GLOB tests CONFIGURE_DEPENDS tests/*.cpp)
if(tests)
	add_executable(kodanuki_tests ${tests} ${sources} ${root}/source/viewer/fly.cpp)
endif()

find_package(SDL3 CONFIG)
if(SDL3_FOUND)
	file(GLOB viewer CONFIGURE_DEPENDS ${root}/source/viewer/*.cpp)
	add_executable(kodanuki_view ${root}/source/viewer.cpp ${viewer} ${sources})
	target_link_libraries(kodanuki_view PRIVATE SDL3::SDL3)
endif()

add_executable(kodanuki_cut_stats tools/cut-stats.cpp tools/spots.cpp ${sources})
add_executable(kodanuki_face_stats tools/face-stats.cpp tools/spots.cpp ${sources})
add_executable(kodanuki_tree_stats tools/tree-stats.cpp ${sources})
add_executable(kodanuki_walk_bench tools/walk-bench.cpp ${sources})
